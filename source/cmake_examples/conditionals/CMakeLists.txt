cmake_minimum_required(VERSION 3.20)
project(multiple LANGUAGES Fortran)

if(UNIX)
	message("This is a Unix build.")
	if(APPLE)
		message("This is Apple build.")
	elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
		message("This is a Linux build.")
	else()
		message("This is a unknown Unix build.")
	endif()
else()
	message("This is not a Unix build.")
	if(WIN32)
		message("This is a Windows build.")
	else()
		message("This is an unknown build.")
	endif()
endif()

add_executable(multiple myprog.f90 mymodule.f90)
